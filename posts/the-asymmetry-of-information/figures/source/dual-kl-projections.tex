\newcommand{\projection}[4]{
  \draw[panel] (0,0) rectangle (7.5,9.1);
  \node[title,text=#3] at (3.75,8.55) {#1};
  \node[smalltext] at (3.75,7.95) {Same target density $p$};
  \begin{scope}[shift={(3.75,6.45)},x=0.32cm,y=3.0cm]
    \draw[maroon,line width=1.1pt] plot file {data/target.dat};
  \end{scope}
  \fill[maroon] (3.75,5.95) circle (2.5pt);
  \node[labeltext,anchor=east] at (3.45,5.95) {$p$};
  \draw[panel,fill=wash] (0.45,0.45) rectangle (7.05,3.15);
  \node[smalltext] at (3.75,0.86) {Gaussian model family $\mathcal M$};
  \draw[vector,#3,shorten >=5pt,shorten <=5pt] (3.75,5.95)--(3.75,2.85);
  \node[labeltext,text=#3,anchor=west,fill=paper,inner sep=5pt] at (3.92,4.55) {$\min\limits_{q\in\mathcal M}$\\[3pt]$\mathrm{KL}(#4)$};
  \fill[#3] (3.75,2.85) circle (2.5pt);
  \node[labeltext,anchor=east,text=#3] at (3.45,2.85) {$q_{#2}^*$};
  \begin{scope}[shift={(3.75,1.28)},x=0.32cm,y=1.55cm]
    \ifnum\pdfstrcmp{#2}{m}=0
      \draw[#3,line width=1.1pt] plot file {data/forward.dat};
    \else
      \draw[#3,line width=1.1pt] plot file {data/reverse.dat};
    \fi
  \end{scope}
}
