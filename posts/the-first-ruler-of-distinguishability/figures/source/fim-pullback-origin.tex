% Schematic embedded patches: the quadratic terms vanish to first order at
% the marked point, so both arrows lie in the actual local tangent plane.
\newcommand{\curvedpatch}[2]{
  \draw[draw=rule,fill=wash,line width=0.65pt]
    plot[domain=-1:1,samples=25] ({1.6*\x-0.75},{0.3*\x+0.65+#1*\x*\x+#2})
    -- plot[domain=-1:1,samples=25] ({1.6+0.75*\x},{0.3-0.65*\x+#1+#2*\x*\x})
    -- plot[domain=1:-1,samples=25] ({1.6*\x+0.75},{0.3*\x-0.65+#1*\x*\x+#2})
    -- plot[domain=1:-1,samples=25] ({-1.6+0.75*\x},{-0.3-0.65*\x+#1+#2*\x*\x}) -- cycle;
  \foreach \v in {-0.5,0,0.5}{
    \draw[rule,line width=0.4pt]
      plot[domain=-1:1,samples=25] ({1.6*\x+0.75*\v},{0.3*\x-0.65*\v+#1*\x*\x+#2*\v*\v});
    \draw[rule,line width=0.4pt]
      plot[domain=-1:1,samples=25] ({1.6*\v+0.75*\x},{0.3*\v-0.65*\x+#1*\v*\v+#2*\x*\x});
  }
  % A small flat patch distinguishes the tangent space from the manifold.
  \draw[draw=muted,dashed,line width=0.6pt,fill=wash,fill opacity=0.88]
    (-1.415,0.13)--(0.665,0.52)--(1.415,-0.13)--(-0.665,-0.52)--cycle;
}
